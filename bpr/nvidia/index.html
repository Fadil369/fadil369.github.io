PHP · lib.php cp_llm
// NVIDIA route chain in cp_llm()
$routes = match ($a['backend']) {
    'nvidia' => ['nvidia', 'nvidia_fallback', 'local'],
};
// NVIDIA API config
'nvidia_url' => 'https://integrate.api.nvidia.com/v1/chat/completions',
'nvidia_token_file' => '/etc/brainsait/nvidia-api.token',
'nvidia_model' => 'nvidia/nemotron-3.5-super-vl-120b-a12b',
'nvidia_fallback_model' => 'nvidia/nemotron-3.5-lightning-30b-a3b',
// Nemotron requires enable_thinking=false, omits cache_prompt
$payload['chat_template_kwargs'] = ['enable_thinking' => false];
